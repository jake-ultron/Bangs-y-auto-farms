local function wipeAlign(root)
    if not root then return end
    for _, n in ipairs({"ToxicAtt","ToxicPos","ToxicOri"}) do
        local c = root:FindFirstChild(n); if c then c:Destroy() end
    end
    pcall(function()
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)
end

local function destroyOldWindows()
    if _G.ToxicFusion_Window then
        pcall(function() _G.ToxicFusion_Window:Destroy() end)
        _G.ToxicFusion_Window = nil
    end
    pcall(function()
        for _, gui in ipairs(CoreGui:GetChildren()) do
            if gui:IsA("ScreenGui") and (gui.Name:find("TOXIC") or gui.Name:find("Toxic")) then
                gui:Destroy()
            end
        end
    end)
    pcall(function()
        local pg = LP:FindFirstChild("PlayerGui")
        if pg then
            for _, gui in ipairs(pg:GetChildren()) do
                if gui:IsA("ScreenGui") and (gui.Name:find("TOXIC") or gui.Name:find("Toxic")) then
                    gui:Destroy()
                end
            end
        end
    end)
    if typeof(gethui) == "function" then
        pcall(function()
            for _, gui in ipairs(gethui():GetChildren()) do
                if gui:IsA("ScreenGui") and (gui.Name:find("TOXIC") or gui.Name:find("Toxic")) then
                    gui:Destroy()
                end
            end
        end)
    end
end

local function FullDestruction()

local function Notificar(t, m, d)
    pcall(function() WindUI:Notify({Title=t, Content=m, Duration=d or 3}) end)

local function ObtenerJugadores()
    local lista = {}; mapaNombres = {}
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP then
            local fmt = p.Name .. "| " .. p.DisplayName
            table.insert(lista, fmt); mapaNombres[fmt] = p.Name
        end
    end
    return lista
end

local function ExtraerNombresReales(val)
    local nombres = {}
    if type(val) == "table" then
        for _, v in ipairs(val) do table.insert(nombres, mapaNombres[v] or v) end
    elseif type(val) == "string" then
        table.insert(nombres, mapaNombres[val] or val)
    end
    return nombres
end

local function matchFilter(p)
    if #PTg == 0 then return true end
    local n, d = p.Name:lower(), p.DisplayName:lower()
    for _, t in ipairs(PTg) do
        if n:find(t,1,true) or d:find(t,1,true) then return true end
    end
    return false
end

local function ObtenerObjetivosValidos()
    local validos = {}
    local function push(p)
        if p and p ~= LP and matchFilter(p)
           and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(validos, p)
        end
    end
    if #TargetNames > 0 then
        for _, name in ipairs(TargetNames) do push(Players:FindFirstChild(name)) end
    else
        for _, p in ipairs(Players:GetPlayers()) do push(p) end
    end
    return validos
end

-- ============================================================
-- ALIGN (Responsiveness 300, MaxForce 2e6)
-- ============================================================
local function ensureAlign()
    local char = LP.Character; if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return nil end
    local att = hrp:FindFirstChild("ToxicAtt")
    if not att then att = Instance.new("Attachment"); att.Name="ToxicAtt"; att.Parent=hrp end

    local ap = hrp:FindFirstChild("ToxicPos")
    if not ap then
        ap = Instance.new("AlignPosition")
        ap.Name="ToxicPos"; ap.Mode=Enum.PositionAlignmentMode.OneAttachment
        ap.Attachment0=att; ap.MaxForce=2e6; ap.Responsiveness=300; ap.Parent=hrp
    else
        ap.MaxForce = 2e6
        ap.Responsiveness = 300
    end

    local ao = hrp:FindFirstChild("ToxicOri")
    if not ao then
        ao = Instance.new("AlignOrientation")
        ao.Name="ToxicOri"; ao.Mode=Enum.OrientationAlignmentMode.OneAttachment
        ao.Attachment0=att; ao.MaxTorque=2e6; ao.Responsiveness=300; ao.Parent=hrp
    else
        ao.MaxTorque = 2e6
        ao.Responsiveness = 300
    end
    return ap, ao
end

local function disableAlign()
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local ap = hrp:FindFirstChild("ToxicPos")
    local ao = hrp:FindFirstChild("ToxicOri")
    if ap then ap.Enabled = false end
    if ao then ao.Enabled = false end
end

-- ============================================================
-- LOOP MAESTRO
-- Prioridad: Bang > Headsit > Orbit
-- ============================================================
_G.Fusion_MainLoop = RunService.Heartbeat:Connect(function(dt)
    local char = LP.Character; if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local myHrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not myHrp or hum.Health <= 0 then return end

    local mode = nil
    if Toggles.Bang then mode = "bang"
    elseif Toggles.Headsit then mode = "headsit"
    elseif Toggles.Orbit then mode = "orbit"
    end
    if not mode then disableAlign(); return end

    local targets = ObtenerObjetivosValidos()
    if #targets == 0 then disableAlign(); return end

    -- Rotación multi-target (excepto en Orbit: se queda fijo al target activo)
    if #targets > 1 and mode ~= "orbit" then
        if tick() - lastSwitchTime >= TargetSwitchInterval then
            currentTargetIndex = currentTargetIndex + 1
            if currentTargetIndex > #targets then currentTargetIndex = 1 end
            activeTarget = targets[currentTargetIndex]
            lastSwitchTime = tick()
        end
        if not activeTarget or not activeTarget.Parent
           or not activeTarget.Character
           or not activeTarget.Character:FindFirstChild("HumanoidRootPart") then
            currentTargetIndex = 1
            activeTarget = targets[1]
            lastSwitchTime = tick()
        end
    else
        -- En orbit o con un solo target: fijar al primero
        if not activeTarget or not activeTarget.Parent
           or not activeTarget.Character
           or not activeTarget.Character:FindFirstChild("HumanoidRootPart") then
            activeTarget = targets[1]
            currentTargetIndex = 1
        end
    end

    local first = activeTarget or targets[1]
    local firstHrp = first and first.Character and first.Character:FindFirstChild("HumanoidRootPart")
    if not firstHrp then return end

    local ap, ao = ensureAlign()
    if not ap or not ao then return end
    ap.Enabled = true; ao.Enabled = true

    local shake = BangShakeEnabled and (math.sin(tick() * BangSpeed * 5) * 0.8) or 0
    local goalPos, lookTarget

    if mode == "bang" then
        goalPos = (firstHrp.CFrame * CFrame.new(OffsetX, OffsetY, OffsetZ + shake)).Position
        lookTarget = firstHrp.Position

    elseif mode == "headsit" then
        local h = first.Character:FindFirstChild("Head") or firstHrp
        goalPos = (h.CFrame * CFrame.new(OffsetX, OffsetY + 0.5 + shake, OffsetZ)).Position
        lookTarget = h.Position

    elseif mode == "orbit" then
        -- Radio mínimo de 3 para no orbitar dentro del target
        local radius = math.abs(OffsetZ)
        if radius < 3 then radius = 3 end

        orbitAngle = orbitAngle + dt * OrbitSpeed

        -- Órbita alrededor del HRP del TARGET ACTIVO (no del centro geométrico)
        local offset = Vector3.new(
            math.sin(orbitAngle) * radius + OffsetX,
            OffsetY,
            math.cos(orbitAngle) * radius
        )
        goalPos = firstHrp.Position + offset
        lookTarget = firstHrp.Position
    end

    if goalPos and lookTarget then
        ap.Position = goalPos
        ao.CFrame = CFrame.lookAt(myHrp.Position, lookTarget)
    end
end)

        else DropdownTargets:SetValue(sel) end
    end)
end

local function forceReset()
    if LP.Character then
        local h = LP.Character:FindFirstChildOfClass("Humanoid")
        if h then h.Health = 0; pcall(function() h:ChangeState(Enum.HumanoidStateType.Dead) end) end
        pcall(function() LP.Character:BreakJoints() end)
    end
end

local function handleAutoReset()
    if not autoResetEnabled or autoResetInterval <= 0 then return end
    local char = LP.Character
    if char then
        local h = char:FindFirstChildOfClass("Humanoid")
        if h and h.Health > 0 and tick() - lastResetTime >= autoResetInterval then
            forceReset(); lastResetTime = tick()
        end
    end
end

_G.Fusion_FarmLoop = RunService.Heartbeat:Connect(handleAutoReset)

    end
end)

local function createMobileButton()
    if _G.Fusion_MobileGui then return end
    local gui = Instance.new("ScreenGui")
    gui.Name = "FusionMobileButton"
    gui.Parent = LP:WaitForChild("PlayerGui")
    gui.ResetOnSpawn = false
    gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    gui.DisplayOrder = 100
    _G.Fusion_MobileGui = gui

    mobileButton = Instance.new("TextButton")
    mobileButton.Size = UDim2.new(0,60,0,45)
    mobileButton.Position = UDim2.fromScale(0.66, 0.65)
    mobileButton.BackgroundColor3 = Toggles.Bang and Color3.fromRGB(37,122,247) or Color3.fromRGB(100,100,100)
    mobileButton.BackgroundTransparency = 0.4
    mobileButton.Text = Toggles.Bang and "BANG\nON" or "BANG\nOFF"
    mobileButton.TextColor3 = Color3.new(1,1,1)
    mobileButton.Font = Enum.Font.GothamBold
    mobileButton.TextSize = 12
    mobileButton.AutoButtonColor = false
    mobileButton.Parent = gui
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0,8); c.Parent = mobileButton

    mobileButton.MouseButton1Click:Connect(function()
        Toggles.Bang = not Toggles.Bang
        lastSwitchTime = 0
        mobileButton.Text = Toggles.Bang and "BANG\nON" or "BANG\nOFF"
        mobileButton.BackgroundColor3 = Toggles.Bang and Color3.fromRGB(37,122,247) or Color3.fromRGB(100,100,100)
        Notificar("BANG", Toggles.Bang and "Activado" or "Desactivado", 2)
    end)

    local dragging, dragInput, dragStart, startPos
    mobileButton.InputBegan:Connect(function(input)
        if (input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch)
           and mobileButtonDragEnabled then
            dragging = true; dragStart = input.Position; startPos = mobileButton.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    mobileButton.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging and mobileButtonDragEnabled then
            local d = input.Position - dragStart
            mobileButton.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function destroyMobileButton()
    if _G.Fusion_MobileGui then
        _G.Fusion_MobileGui:Destroy(); _G.Fusion_MobileGui = nil; mobileButton = nil
    end
end
