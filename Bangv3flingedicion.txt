--// HELPER FUNCTIONS PARA MULTI DROPDOWN
local function Notificar(titulo, mensaje, duracion)
    pcall(function()
        WindUI:Notify({
            Title = titulo,
            Content = mensaje,
            Duration = duracion or 3
        })
    end)
end

local function ObtenerJugadores()
    local lista = {}
    mapaNombres = {}
    for _, p in pairs(game.Players:GetPlayers()) do 
        if p ~= game.Players.LocalPlayer then 
            local fmt = p.Name .. "| " .. p.DisplayName
            table.insert(lista, fmt)
            mapaNombres[fmt] = p.Name
        end
    end
    return lista
end

local function ExtraerNombresReales(val)
    local nombres = {}
    if type(val) == "table" then
        for _, v in ipairs(val) do
            table.insert(nombres, mapaNombres[v] or v)
        end
    elseif type(val) == "string" then
        table.insert(nombres, mapaNombres[val] or val)
    end
    return nombres
end

local function ObtenerObjetivosValidos()
    local validos = {}
    for _, name in ipairs(TargetNames) do
        local p = game.Players:FindFirstChild(name)
        if p and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(validos, p)
        end
    end
    return validos
end

--// MULTI DROPDOWN DE TARGETS
local DropdownTargets = TrollTab:Dropdown({
    Title = "Fijar Objetivos (Targets Multi)",
    Desc = "Selecciona uno o varios jugadores para las acciones.",
    Values = ObtenerJugadores(),
    Value = nil,
    AllowNone = true,
    Multi = true,
    Callback = function(val)
        if refrescandoLista then return end
        
        if val == nil then
            TargetNames = {}
        else
            TargetNames = ExtraerNombresReales(val)
        end
        
        if #TargetNames == 0 then
            Notificar("✅ Objetivos Limpios", "Sin objetivos seleccionados.", 3)
        else
            local nombres = table.concat(TargetNames, ", ")
            if #nombres > 50 then nombres = string.sub(nombres, 1, 50).."..." end
            Notificar("🎯 Objetivos Fijados ("..#TargetNames..")", nombres, 3)
        end
    end
})

-- Sincronizar Dropdown
local function SincronizarDropdown()
    if not DropdownTargets then return end
    local seleccionadosFormateados = {}
    for fmt, nombreReal in pairs(mapaNombres) do
        for _, t in ipairs(TargetNames) do
            if t == nombreReal then
                table.insert(seleccionadosFormateados, fmt)
                break
            end
        end
    end
    pcall(function()
        if #seleccionadosFormateados == 0 then
            DropdownTargets:SetValue(nil)
        else
            DropdownTargets:SetValue(seleccionadosFormateados)
        end
    end)
end

        if value then
            local teleportPosition = Vector3.new(5000, 1000, 5000)
            local baseplateSize = Vector3.new(100, 1, 100)
            local baseplatePosition = teleportPosition - Vector3.new(0, 5, 0)

            SafeSpotBaseplate = Instance.new("Part")
            SafeSpotBaseplate.Size = baseplateSize
            SafeSpotBaseplate.Position = baseplatePosition
            SafeSpotBaseplate.Anchored = true
            SafeSpotBaseplate.BrickColor = BrickColor.new("Earth green")
            SafeSpotBaseplate.Parent = workspace

            local playerGui = player:WaitForChild("PlayerGui")
            SafeSpotGui = Instance.new("ScreenGui")
            SafeSpotGui.Name = "TeleportGui"
            SafeSpotGui.ResetOnSpawn = false
            SafeSpotGui.Parent = playerGui

            local teleportButton = Instance.new("TextButton")
            teleportButton.Size = UDim2.new(0, 50, 0, 50)
            teleportButton.Position = UDim2.new(0, 150, 0, 10)
            teleportButton.BackgroundColor3 = Color3.fromRGB(0, 200, 0)
            teleportButton.Text = "TP"
            teleportButton.TextColor3 = Color3.fromRGB(255, 255, 255)
            teleportButton.Font = Enum.Font.GothamBold
            teleportButton.TextSize = 20
            teleportButton.Parent = SafeSpotGui
            
            local corner = Instance.new("UICorner")
            corner.CornerRadius = UDim.new(0, 8)
            corner.Parent = teleportButton

            local dragging, dragInput, dragStart, startPos

            teleportButton.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                    dragStart = input.Position
                    startPos = teleportButton.Position
                    
                    input.Changed:Connect(function()
                        if input.UserInputState == Enum.UserInputState.End then
                            dragging = false
                        end
                    end)
                end
            end)

            teleportButton.InputChanged:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                    dragInput = input
                end
            end)

            UserInputService.InputChanged:Connect(function(input)
                if input == dragInput and dragging then
                    local delta = input.Position - dragStart
                    teleportButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
                end
            end)

            teleportButton.MouseButton1Click:Connect(function()
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    local humanoidRootPart = player.Character.HumanoidRootPart

                    if not teleported then
                        originalPosition = humanoidRootPart.Position
                        humanoidRootPart.CFrame = CFrame.new(teleportPosition)
                        teleported = true
                        teleportButton.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
                        teleportButton.Text = "BACK"
                    else
                        if originalPosition then
                            humanoidRootPart.CFrame = CFrame.new(originalPosition)
                        end
                        teleported = false
                        teleportButton.BackgroundColor3 = Color3.fromRGB(0, 200, 0)
                        teleportButton.Text = "TP"
                    end
                end
            end)

        else
            if SafeSpotGui then
                SafeSpotGui:Destroy()
                SafeSpotGui = nil
            end
            if SafeSpotBaseplate then
                SafeSpotBaseplate:Destroy()
                SafeSpotBaseplate = nil
            end
            
            if teleported and player and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                if originalPosition then
                    player.Character.HumanoidRootPart.CFrame = CFrame.new(originalPosition)
                end
                teleported = false
            end
        end
    end,
})
