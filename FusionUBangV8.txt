local function containers()
    local c = {CoreGui}
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then table.insert(c, pg) end
    if typeof(gethui) == "function" then pcall(function() table.insert(c, gethui()) end) end
    return c
end
local function isToxic(g)
    if not g or not g:IsA("ScreenGui") then return false end
    for _, d in ipairs(g:GetDescendants()) do
        if d:IsA("TextLabel") and d.Text == "TOXIC HS FUSION" then return true end
    end
    return false
end
local function killOld(exc)
    for _, ct in ipairs(containers()) do
        if ct then pcall(function()
            for _, g in ipairs(ct:GetChildren()) do
                if g ~= exc and isToxic(g) then g:Destroy() end
            end
        end) end
    end
end
local function killAll()
    for _, k in ipairs({"Fusion_MainLoop", "Fusion_GrabFling"}) do
        local c = _G[k]
        if c then pcall(function() c:Disconnect() end); _G[k] = nil end
    end
    if _G.Fusion_Toggles then
        for k in pairs(_G.Fusion_Toggles) do _G.Fusion_Toggles[k] = false end
    end
end
killAll(); killOld()

local PhysPitch, PhysRoll = -90, 0
local PhysSpin, PhysYawMode = 60, 1

local BangSpeed, BangShakeEnabled, spinAcc = 1, false, 0

-- ===== HELPERS =====
local function Notif(t, m, d)
    pcall(function() WindUI:Notify({Title=t, Content=m, Duration=d or 3}) end)
end
local function getPlayers()
    local lista = {}
    mapaNombres = {}
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP then
            local fmt = p.Name .. "| " .. p.DisplayName
            table.insert(lista, fmt)
            mapaNombres[fmt] = p.Name
        end
    end
    return lista
end
local function nombresReales(val)
    local r = {}
    if type(val) == "table" then
        for _, v in ipairs(val) do table.insert(r, mapaNombres[v] or v) end
    elseif type(val) == "string" then
        table.insert(r, mapaNombres[val] or val)
    end
    return r
end
local function matchFilter(p)
    if #PTg == 0 then return true end
    for _, t in ipairs(PTg) do
        if p.Name:lower():find(t, 1, true) or p.DisplayName:lower():find(t, 1, true) then return true end
    end
    return false
end
local function getTargets()
    local r = {}
    local function push(p)
        if p and p ~= LP and matchFilter(p) and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(r, p)
        end
    end
    if #TargetNames > 0 then
        for _, n in ipairs(TargetNames) do push(Players:FindFirstChild(n)) end
    else
        for _, p in ipairs(Players:GetPlayers()) do push(p) end
    end
    return r
end
local function wasdDir()
    local cam = workspace.CurrentCamera
    if not cam then return Vector3.zero end
    local d = Vector3.zero
    if UIS:IsKeyDown(Enum.KeyCode.W) then d = d + cam.CFrame.LookVector end
    if UIS:IsKeyDown(Enum.KeyCode.S) then d = d - cam.CFrame.LookVector end
    if UIS:IsKeyDown(Enum.KeyCode.A) then d = d - cam.CFrame.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.D) then d = d + cam.CFrame.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.Space) then d = d + Vector3.yAxis end
    return d
end
local function pickTarget(tgts)
    if #tgts > 1 then
        if tick() - lastSwitchTime >= TargetSwitchInterval then
            currentTargetIndex = currentTargetIndex % #tgts + 1
            activeTarget = tgts[currentTargetIndex]
            lastSwitchTime = tick()
        end
        if not activeTarget or not activeTarget.Parent then
            currentTargetIndex = 1
            activeTarget = tgts[1]
        end
        return activeTarget
    end
    return tgts[1]
end

-- ===== MAIN LOOP (Direct CFrame) =====
_G.Fusion_MainLoop = RunService.RenderStepped:Connect(function(dt)
    local char = LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then return end
    local mode = nil
    if Toggles.Bang then mode = "bang"
    elseif Toggles.Headsit then mode = "headsit"
    elseif Toggles.Orbit then mode = "orbit" end
    if not mode then
        if hum.PlatformStand then hum.PlatformStand = false; hum.AutoRotate = true end
        frozenCF = nil
        return
    end
    if not hum.PlatformStand then
        hum.PlatformStand = true
        hum.AutoRotate = false
        pcall(function() for _, tr in ipairs(hum:GetPlayingAnimationTracks()) do tr:Stop() end end)
    end
    local tgts = getTargets()
    if #tgts == 0 then return end
    local tgt = pickTarget(tgts)
    local thrp = tgt and tgt.Character and tgt.Character:FindFirstChild("HumanoidRootPart")
    if not thrp then return end
    if PhysYawMode == 3 then spinAcc = spinAcc + PhysSpin * dt end
    local shake = BangShakeEnabled and math.sin(tick() * BangSpeed * 5) * 0.8 or 0
    local yaw
    if PhysYawMode == 1 then yaw = math.rad(thrp.Orientation.Y)
    elseif PhysYawMode == 2 then yaw = math.rad(hrp.Orientation.Y)
    else yaw = math.rad(spinAcc) end
    local base = CFrame.Angles(0, yaw, 0)
    local rot = base * CFrame.Angles(math.rad(PhysPitch), 0, math.rad(PhysRoll))
    local pos
    if mode == "bang" then
        pos = thrp.Position + base:VectorToWorldSpace(Vector3.new(OffsetX, OffsetY + shake, OffsetZ))
    elseif mode == "headsit" then
        local h = tgt.Character:FindFirstChild("Head") or thrp
        pos = h.Position + base:VectorToWorldSpace(Vector3.new(OffsetX, OffsetY + 0.5 + shake, OffsetZ))
    else
        local r = math.max(math.abs(OffsetZ), 3)
        orbitAngle = orbitAngle + dt * OrbitSpeed
        pos = thrp.Position + Vector3.new(math.sin(orbitAngle) * r + OffsetX, OffsetY, math.cos(orbitAngle) * r)
    end
    local cf = (holdMode and frozenCF) or (CFrame.new(pos) * rot)
    if holdMode and not frozenCF then frozenCF = cf end
    if moveMode then
        local d = wasdDir()
        if d.Magnitude > 0.01 then moveOffset = moveOffset + d.Unit * strafeSpeed * dt end
    end
    hrp.CFrame = cf + moveOffset
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hum.MoveDirection = Vector3.zero
end)
print("[TOXIC] Main loop OK")

-- ===== ANTI-FLING GRAB =====
_G.Fusion_GrabFling = RunService.Heartbeat:Connect(function()
    if not antiFlingGrab then return end
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local welded = false
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") then
            local a, b = d.Part0, d.Part1
            if a and b then
                local other = nil
                if a:IsDescendantOf(char) and not b:IsDescendantOf(char) then other = b
                elseif b:IsDescendantOf(char) and not a:IsDescendantOf(char) then other = a end
                if other then welded = true; break end
            end
        end
    end
    if welded then
        if not grabActive then grabActive = true; grabDampUntil = os.clock() + grabDamp end
        if os.clock() < grabDampUntil then
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        else
            local v = hrp.AssemblyLinearVelocity
            if v.Magnitude > maxGrabSpeed then hrp.AssemblyLinearVelocity = v.Unit * maxGrabSpeed end
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    else
        grabActive = false
    end
end)
print("[TOXIC] Grab fling OK")

    if not dd then return end

local function ddRef(dd, lista)
    if not dd then return end
    pcall(function()
        if dd.Refresh then dd:Refresh(lista)
        elseif dd.SetOptions then dd:SetOptions(lista) end
    end)
end
