--TOXIC FUSION V10 (DIRECT CF)
local Players=game:GetService("Players")
local RunService=game:GetService("RunService")
local UIS=game:GetService("UserInputService")
local CoreGui=game:GetService("CoreGui")
local LP=Players.LocalPlayer
local COOLDOWN=2.0
if _G.ToxicFusion_LastStart and (tick()-_G.ToxicFusion_LastStart)<COOLDOWN then warn(string.format("[TOXIC] Cooldown %.1fs",COOLDOWN-(tick()-_G.ToxicFusion_LastStart)));return end
_G.ToxicFusion_LastStart=tick()
local function getAllContainers()
  local c={CoreGui}
  local pg=LP:FindFirstChild("PlayerGui");if pg then table.insert(c,pg) end
  if typeof(gethui)=="function" then pcall(function() table.insert(c,gethui()) end) end
  return c
end
local function isToxicScreen(gui)
  if not gui or not gui:IsA("ScreenGui") then return false end
  for _,d in ipairs(gui:GetDescendants()) do if d:IsA("TextLabel") and d.Text=="TOXIC HS FUSION" then return true end end
  return false
end
local function destroyOldWindows(exc)
  for _,ct in ipairs(getAllContainers()) do if ct then pcall(function()
    for _,g in ipairs(ct:GetChildren()) do if g~=exc and isToxicScreen(g) then g:Destroy() end end
  end) end end
end
local function wipeAlign(root)
  if not root then return end
  for _,n in ipairs({"ToxicAtt","ToxicPos","ToxicOri"}) do local c=root:FindFirstChild(n);if c then c:Destroy() end end
  pcall(function() root.AssemblyLinearVelocity=Vector3.zero;root.AssemblyAngularVelocity=Vector3.zero end)
end
local function FullDestruction()
  for _,k in ipairs({"Fusion_FarmLoop","Fusion_SpawnConn","Fusion_WalkingFling","Fusion_AntiFling","Fusion_MainLoop","Fusion_GrabFling"}) do if _G[k] then _G[k]:Disconnect();_G[k]=nil end end
  if _G.Fusion_MobileGui then pcall(function() _G.Fusion_MobileGui:Destroy() end);_G.Fusion_MobileGui=nil end
  if _G.Fusion_SafeSpotCleanup then pcall(_G.Fusion_SafeSpotCleanup);_G.Fusion_SafeSpotCleanup=nil end
  _G.Fusion_AutoRefresh=nil
  if _G.Fusion_Toggles then for k in pairs(_G.Fusion_Toggles) do _G.Fusion_Toggles[k]=false end end
  if LP and LP.Character then wipeAlign(LP.Character:FindFirstChild("HumanoidRootPart")) end
end
FullDestruction();destroyOldWindows()
