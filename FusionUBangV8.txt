local function getContainers()
    local c = {CoreGui}
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then table.insert(c, pg) end
    if typeof(gethui) == "function" then
        pcall(function() table.insert(c, gethui()) end)
    end
    return c
end

local function isToxicScreen(gui)
    if not gui or not gui:IsA("ScreenGui") then return false end
    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("TextLabel") and d.Text == "TOXIC HS FUSION" then
            return true
        end
    end
    return false
end

local function destroyOld(exc)
    for _, ct in ipairs(getContainers()) do
        if ct then
            pcall(function()
                for _, g in ipairs(ct:GetChildren()) do
                    if g ~= exc and isToxicScreen(g) then
                        g:Destroy()
                    end
                end
            end)
        end
    end
end

local function fullDestroy()
    for _, k in ipairs({"Fusion_MainLoop", "Fusion_GrabFling", "Fusion_FarmLoop", "Fusion_SpawnConn"}) do
        local c = _G[k]
        if c then
            pcall(function() c:Disconnect() end)
            _G[k] = nil
        end
    end
    if _G.Fusion_MobileGui then
        pcall(function() _G.Fusion_MobileGui:Destroy() end)
        _G.Fusion_MobileGui = nil
    end
    if _G.Fusion_Toggles then
        for k in pairs(_G.Fusion_Toggles) do _G.Fusion_Toggles[k] = false end
    end
end

local function Notificar(title, msg, dur)

local function ObtenerJugadores()
    local lista = {}
    mapaNombres = {}
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP then
            local fmt = p.Name .. "| " .. p.DisplayName
            table.insert(lista, fmt)
            mapaNombres[fmt] = p.Name
        end
    end
    return lista
end

local function ExtraerNombresReales(val)
    local nombres = {}
    if type(val) == "table" then
        for _, v in ipairs(val) do
            table.insert(nombres, mapaNombres[v] or v)
        end
    elseif type(val) == "string" then
        table.insert(nombres, mapaNombres[val] or val)
    end
    return nombres
end

local function matchFilter(p)
    if #PTg == 0 then return true end
    local n, d = p.Name:lower(), p.DisplayName:lower()
    for _, t in ipairs(PTg) do
        if n:find(t, 1, true) or d:find(t, 1, true) then return true end
    end
    return false
end

local function ObtenerObjetivosValidos()
    local validos = {}
    local function push(p)
        if p and p ~= LP and matchFilter(p) and p.Character
           and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(validos, p)
        end
    end
    if #TargetNames > 0 then
        for _, name in ipairs(TargetNames) do push(Players:FindFirstChild(name)) end
    else
        for _, p in ipairs(Players:GetPlayers()) do push(p) end
    end
    return validos
end

local function wasdDir()
    local cam = workspace.CurrentCamera
    if not cam then return Vector3.zero end
    local dir = Vector3.zero
    if UIS:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
    if UIS:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
    if UIS:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.yAxis end
    if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.yAxis end
    return dir
end

local function pickActiveTarget(targets)
    if #targets > 1 then
        if tick() - lastSwitchTime >= TargetSwitchInterval then
            currentTargetIndex = currentTargetIndex + 1
            if currentTargetIndex > #targets then currentTargetIndex = 1 end
            activeTarget = targets[currentTargetIndex]
            lastSwitchTime = tick()
        end
        if not activeTarget or not activeTarget.Parent or not activeTarget.Character
           or not activeTarget.Character:FindFirstChild("HumanoidRootPart") then
            currentTargetIndex = 1
            activeTarget = targets[1]
            lastSwitchTime = tick()
        end
        return activeTarget
    end
    activeTarget = targets[1]
    currentTargetIndex = 1
    return activeTarget
end

_G.Fusion_MainLoop = RunService.RenderStepped:Connect(function(dt)
    local char = LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then return end
    local mode = nil
    if Toggles.Bang then mode = "bang"
    elseif Toggles.Headsit then mode = "headsit"
    elseif Toggles.Orbit then mode = "orbit" end
    if not mode then
        if hum.PlatformStand then hum.PlatformStand = false; hum.AutoRotate = true end
        if frozenCF then frozenCF = nil end
        return
    end
    if not hum.PlatformStand then
        hum.PlatformStand = true
        hum.AutoRotate = false
        pcall(function()
            for _, tr in ipairs(hum:GetPlayingAnimationTracks()) do tr:Stop() end
        end)
    end
    local targets = ObtenerObjetivosValidos()
    if #targets == 0 then return end
    local first = pickActiveTarget(targets)
    local firstHrp = first and first.Character and first.Character:FindFirstChild("HumanoidRootPart")
    if not firstHrp then return end
    if PhysYawMode == 3 then spinAcc = spinAcc + PhysSpin * dt end
    local shake = BangShakeEnabled and (math.sin(tick() * BangSpeed * 5) * 0.8) or 0
    local yaw
    if PhysYawMode == 1 then yaw = math.rad(firstHrp.Orientation.Y)
    elseif PhysYawMode == 2 then yaw = math.rad(hrp.Orientation.Y)
    else yaw = math.rad(spinAcc) end
    local base = CFrame.Angles(0, yaw, 0)
    local rot = base * CFrame.Angles(math.rad(PhysPitch), 0, math.rad(PhysRoll))
    local pos
    if mode == "bang" then
        pos = firstHrp.Position + base:VectorToWorldSpace(Vector3.new(OffsetX, OffsetY + shake, OffsetZ))
    elseif mode == "headsit" then
        local h = first.Character:FindFirstChild("Head") or firstHrp
        pos = h.Position + base:VectorToWorldSpace(Vector3.new(OffsetX, OffsetY + 0.5 + shake, OffsetZ))
    else
        local radius = math.abs(OffsetZ)
        if radius < 3 then radius = 3 end
        orbitAngle = orbitAngle + dt * OrbitSpeed
        pos = firstHrp.Position + Vector3.new(math.sin(orbitAngle) * radius + OffsetX, OffsetY, math.cos(orbitAngle) * radius)
    end
    local cf = holdMode and frozenCF or (CFrame.new(pos) * rot)
    if holdMode and not frozenCF then frozenCF = cf end
    if moveMode then
        local dir = wasdDir()
        if dir.Magnitude > 0.01 then
            moveOffset = moveOffset + dir.Unit * strafeSpeed * dt
        end
    end
    hrp.CFrame = cf + moveOffset
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hum.MoveDirection = Vector3.zero
end)

-- ===== ANTI-FLING ON GRAB =====
_G.Fusion_GrabFling = RunService.Heartbeat:Connect(function()
    if not antiFlingGrab then return end
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local welded = false
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") then
            local a, b = d.Part0, d.Part1
            if a and b then
                local other = nil
                if a:IsDescendantOf(char) and not b:IsDescendantOf(char) then other = b
                elseif b:IsDescendantOf(char) and not a:IsDescendantOf(char) then other = a end
                if other then welded = true; break end
            end
        end
    end
    if welded then
        if not grabActive then
            grabActive = true
            grabDampUntil = os.clock() + grabDamp
        end
        if os.clock() < grabDampUntil then
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        else
            local v = hrp.AssemblyLinearVelocity
            if v.Magnitude > maxGrabSpeed then
                hrp.AssemblyLinearVelocity = v.Unit * maxGrabSpeed
            end
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    else
        grabActive = false
    end
end)

local function forceReset()
    if LP.Character then
        local h = LP.Character:FindFirstChildOfClass("Humanoid")
        if h then
            h.Health = 0
            pcall(function() h:ChangeState(Enum.HumanoidStateType.Dead) end)
        end
        pcall(function() LP.Character:BreakJoints() end)
    end
end

local function handleAutoReset()
    if not autoResetEnabled or autoResetInterval <= 0 then return end
    local char = LP.Character
    if char then
        local h = char:FindFirstChildOfClass("Humanoid")
        if h and h.Health > 0 and tick() - lastResetTime >= autoResetInterval then
            forceReset()
            lastResetTime = tick()
        end
    end
end
