local function isThisHubScreen(gui)
    if not gui or not gui:IsA("ScreenGui") then return false end
    if gui.Name == "FusionMobileButton" or gui.Name == HUB_ID then return true end
    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("TextLabel") and (d.Text == HUB_NAME or d.Text == "TOXIC X FUSION") then return true end
    end
    return false
end

local function destroyOldWindows(exceptGui)
    for _, container in ipairs(getAllContainers()) do
        if container then
            pcall(function()
                for _, gui in ipairs(container:GetChildren()) do
                    if gui ~= exceptGui and isThisHubScreen(gui) then gui:Destroy() end
                end
            end)
        end
    end
end

local NOTIF_DELAY = 2.0; local lastNotif, pendingNotif = 0, nil
local function mostrarNotif(t, m, d) lastNotif = tick(); pcall(function() Window:Notify({Title=t, Content=m, Duration=d or 3}) end) end
local function Notificar(t, m, d)
    local espera = NOTIF_DELAY - (tick() - lastNotif)
    if pendingNotif then pendingNotif = {t, m, d}; return end
    if espera <= 0 then mostrarNotif(t, m, d); return end
    pendingNotif = {t, m, d}; task.delay(espera, function() local p = pendingNotif; pendingNotif = nil; if p then mostrarNotif(p[1], p[2], p[3]) end end)
end
local function listaJugadores()
    local n = {}
    for _, p in ipairs(Players:GetPlayers()) do if p ~= LP then table.insert(n, p.Name .. "| " .. p.DisplayName) end end
    table.sort(n)
    return n
end
local function listaObjetivos()
    local t = {}
    for k in pairs(targetSet) do local p = Players:FindFirstChild(k); if p then table.insert(t, p.Name .. "| " .. p.DisplayName) end end
    return t
end
local function matchFilter(p)
    if #PTg == 0 then return true end
    local n, d = p.Name:lower(), p.DisplayName:lower()
    for _, t in ipairs(PTg) do if n:find(t,1,true) or d:find(t,1,true) then return true end end
    return false
end
local function ObtenerObjetivosValidos()
    local validos = {}
    local function push(p) if p and p ~= LP and matchFilter(p) and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then table.insert(validos, p) end end
    if next(targetSet) then for name in pairs(targetSet) do push(Players:FindFirstChild(name)) end
    else for _, p in ipairs(Players:GetPlayers()) do push(p) end end
    return validos
end
local function wasdDir()
    local cam = workspace.CurrentCamera; if not cam then return Vector3.zero end
    local dir = Vector3.zero
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cam.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cam.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cam.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cam.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end
    return dir
end
local function weldedPlayer()
    local char = LP.Character; if not char then return nil end
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") or d:IsA("Snap") then
            local a, b = d.Part0, d.Part1
            if a and b then
                local other = nil
                if a:IsDescendantOf(char) and not b:IsDescendantOf(char) then other = b
                elseif b:IsDescendantOf(char) and not a:IsDescendantOf(char) then other = a end
                if other then local model = other:FindFirstAncestorOfClass("Model"); local plr = model and Players:GetPlayerFromCharacter(model); if plr and plr ~= LP then return plr end end
            end
        end
    end
    return nil
end
local function pickActiveTarget(targets)
    if AlwaysClosest then
        local myHRP = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if not myHRP then return targets[1] end
        local best, bd = nil, math.huge
        for _, p in ipairs(targets) do local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart"); if hrp then local d = (hrp.Position - myHRP.Position).Magnitude; if d < bd then bd = d; best = p end end end
        return best or targets[1]
    end
    if #targets > 1 then
        if tick() - lastSwitchTime >= TargetSwitchInterval then currentTargetIndex = currentTargetIndex + 1; if currentTargetIndex > #targets then currentTargetIndex = 1 end; activeTarget = targets[currentTargetIndex]; lastSwitchTime = tick() end
        if not activeTarget or not activeTarget.Parent or not activeTarget.Character or not activeTarget.Character:FindFirstChild("HumanoidRootPart") then currentTargetIndex = 1; activeTarget = targets[1]; lastSwitchTime = tick() end
        return activeTarget
    end
    activeTarget = targets[1]; currentTargetIndex = 1; return activeTarget
end

-- ===== STRAFE STATE =====
local strafeState = nil
local function strafeEnable()
    local char = LP.Character; local hum = char and char:FindFirstChildOfClass("Humanoid"); local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end
    freeYaw = math.rad(hrp.Orientation.Y); strafeState = { anchored = hrp.Anchored, platform = hum.PlatformStand, autorotate = hum.AutoRotate }
    hum.PlatformStand = true; hum.AutoRotate = false; if not SyncMode then hrp.Anchored = true end
    pcall(function() for _, tr in ipairs(hum:GetPlayingAnimationTracks()) do tr:Stop() end end)
    moveOffset = Vector3.zero; frozenCF = nil; strafeGrabCF = nil
end
local function strafeDisable()
    local char = LP.Character; local hum = char and char:FindFirstChildOfClass("Humanoid"); local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if strafeState and hum and hrp then hrp.Anchored = strafeState.anchored; hum.PlatformStand = strafeState.platform; hum.AutoRotate = strafeState.autorotate end
    strafeState = nil; frozenCF = nil; strafeGrabCF = nil; moveOffset = Vector3.zero
end

-- ===== LOOP MAESTRO =====
_G.Fusion_MainLoop = RunService.RenderStepped:Connect(function(dt)
    if isDead then return end
    local char = LP.Character; if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid"); local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then return end
    
    local mode = nil
    if Toggles.Strafe then mode = "strafe" elseif Toggles.Headsit then mode = "headsit" elseif Toggles.Orbit then mode = "orbit" end
    if mode ~= currentMode then if mode then strafeEnable() else strafeDisable() end; currentMode = mode end
    if not mode then return end
    if not hum.PlatformStand then hum.PlatformStand = true; hum.AutoRotate = false end
    if SyncMode and hrp.Anchored then hrp.Anchored = false end
    if not SyncMode and not hrp.Anchored then hrp.Anchored = true end
    
    if PhysYawMode == 3 then spinAcc = spinAcc + PhysSpin * dt end
    local targets = ObtenerObjetivosValidos(); if #targets == 0 then return end
    local first = pickActiveTarget(targets)
    local firstHead = first and first.Character and (first.Character:FindFirstChild("Head") or first.Character:FindFirstChild("HumanoidRootPart"))
    local firstHrp = first and first.Character and first.Character:FindFirstChild("HumanoidRootPart")
    if not firstHead or not firstHrp then return end
    
    local welded = weldedPlayer()
    
    if welded and ResetOnGrab and not isDead then
        local savePos = hrp.Position
        isDead = true
        hum.Health = 0
        task.spawn(function()
            LP.CharacterAdded:Wait()
            task.wait(ResetGrabTime)
            local newHrp = LP.Character:WaitForChild("HumanoidRootPart", 5)
            if newHrp then newHrp.CFrame = CFrame.new(savePos) end
            task.wait(0.5) -- Pequeño delay para evitar bucle si te agarran al instante
            isDead = false
            if Toggles.Strafe or Toggles.Headsit or Toggles.Orbit then Notificar("AUTO GRAB", "Regresando a posición...", 3) end
        end)
        return
    end
    
    if AntiFlingOnGrab and welded then
        if not grabActive then grabActive = true; grabDampUntil = os.clock() + GrabDamp end
        if os.clock() < grabDampUntil then hrp.AssemblyLinearVelocity = Vector3.zero; hrp.AssemblyAngularVelocity = Vector3.zero
        else local v = hrp.AssemblyLinearVelocity; if v.Magnitude > MaxGrabSpeed then hrp.AssemblyLinearVelocity = v.Unit * MaxGrabSpeed end; hrp.AssemblyAngularVelocity = Vector3.zero end
    else grabActive = false end
    
    if mode == "strafe" and welded and StabOnGrab then
        if not strafeGrabCF then strafeGrabCF = hrp.CFrame end
        if WASDMoveEnabled then local dir = wasdDir(); if dir.Magnitude > 0.01 then strafeGrabCF = CFrame.new(strafeGrabCF.Position + dir.Unit * StrafeSpeed * dt) * strafeGrabCF.Rotation end end
        hrp.CFrame = strafeGrabCF
        if SyncMode then hrp.AssemblyLinearVelocity = Vector3.zero; hrp.AssemblyAngularVelocity = Vector3.zero; hum.MoveDirection = Vector3.zero end
        return
    end
    strafeGrabCF = nil
    
    if WASDMoveEnabled then local dir = wasdDir(); if dir.Magnitude > 0.01 then moveOffset += dir.Unit * StrafeSpeed * dt end end
    
    local cf
    if HoldMode and frozenCF then cf = frozenCF
    else
        local yaw
        if PhysYawMode == 1 then yaw = math.rad(firstHrp.Orientation.Y) elseif PhysYawMode == 2 then yaw = freeYaw else yaw = math.rad(spinAcc) end
        local base = CFrame.Angles(0, yaw, 0); local shake = BangShakeEnabled and (math.sin(tick() * BangSpeed * 5) * 0.8) or 0
        if mode == "orbit" then
            local radius = math.abs(OffsetZ); if radius < 3 then radius = 3 end
            orbitAngle = orbitAngle + dt * OrbitSpeed
            local off = Vector3.new(math.sin(orbitAngle) * radius + OffsetX, OffsetY + shake, math.cos(orbitAngle) * radius)
            cf = CFrame.new(firstHrp.Position + off) * base * CFrame.Angles(math.rad(PhysPitch), 0, math.rad(PhysRoll))
        else
            local anchor = firstHead.Position; if mode == "strafe" then anchor = firstHrp.Position end
            local pos = anchor + Vector3.new(0, OffsetY + shake, 0) + base.RightVector * OffsetX - base.LookVector * OffsetZ
            cf = CFrame.new(pos) * base * CFrame.Angles(math.rad(PhysPitch), 0, math.rad(PhysRoll))
        end
        if HoldMode then frozenCF = cf end
    end
    
    if cf then
        if PhysSpeed >= 10 then hrp.CFrame = cf + moveOffset
        else hrp.CFrame = hrp.CFrame:Lerp(cf + moveOffset, math.clamp(PhysSpeed * dt, 0, 1)) end
        if SyncMode then hrp.AssemblyLinearVelocity = Vector3.zero; hrp.AssemblyAngularVelocity = Vector3.zero; hum.MoveDirection = Vector3.zero end
    end
end)
