local function Notificar(titulo, mensaje, duracion)
    pcall(function()
        WindUI:Notify({Title = titulo, Content = mensaje, Duration = duracion or 3})
    end)
end

local function ObtenerJugadores()
    local lista = {}
    mapaNombres = {}
    for _, p in pairs(Players:GetPlayers()) do 
        if p ~= LP then 
            local fmt = p.Name .. "| " .. p.DisplayName
            table.insert(lista, fmt)
            mapaNombres[fmt] = p.Name
        end
    end
    return lista
end

local function ExtraerNombresReales(val)
    local nombres = {}
    if type(val) == "table" then
        for _, v in ipairs(val) do
            table.insert(nombres, mapaNombres[v] or v)
        end
    elseif type(val) == "string" then
        table.insert(nombres, mapaNombres[val] or val)
    end
    return nombres
end

local function matchFilter(player)
    if #PTg == 0 then return true end
    local n, d = player.Name:lower(), player.DisplayName:lower()
    for _, t in ipairs(PTg) do
        if n:find(t, 1, true) or d:find(t, 1, true) then return true end
    end
    return false
end

local function ObtenerObjetivosValidos()
    local validos = {}
    if #TargetNames > 0 then
        for _, name in ipairs(TargetNames) do
            local p = Players:FindFirstChild(name)
            if p and p ~= LP and matchFilter(p)
               and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                table.insert(validos, p)
            end
        end
    else
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and matchFilter(p)
               and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                table.insert(validos, p)
            end
        end
    end
    return validos
end

-- Align physics helpers
local function getOrCreateAlign(root)
    local att = root:FindFirstChild("ToxicAtt")
    if not att then
        att = Instance.new("Attachment")
        att.Name = "ToxicAtt"
        att.Parent = root
    end
    
    local ap = root:FindFirstChild("ToxicPos")
    if not ap then
        ap = Instance.new("AlignPosition")
        ap.Name = "ToxicPos"
        ap.Mode = Enum.PositionAlignmentMode.OneAttachment
        ap.Attachment0 = att
        ap.MaxForce = 1000000
        ap.Responsiveness = 150
        ap.Parent = root
    end
    
    local ao = root:FindFirstChild("ToxicOri")
    if not ao then
        ao = Instance.new("AlignOrientation")
        ao.Name = "ToxicOri"
        ao.Mode = Enum.OrientationAlignmentMode.OneAttachment
        ao.Attachment0 = att
        ao.MaxTorque = 1000000
        ao.Responsiveness = 150
        ao.Parent = root
    end
    
    return att, ap, ao
end

local function clearAlign(root)
    if not root then return end
    for _, n in ipairs({"ToxicAtt", "ToxicPos", "ToxicOri"}) do
        local c = root:FindFirstChild(n)
        if c then c:Destroy() end
    end
end
